local function OpenCase(name, count)
    local Events = ReplicatedStorage:FindFirstChild("Events")
    if not Events then return false, "Events not found" end
    local openFunc = Events:FindFirstChild("OpenCase")
    if not openFunc then return false, "OpenCase not found" end
    return openFunc:InvokeServer(name, count)
end

local function addCorner(p, r)
    local v = Instance.new("UICorner", p)
    v.CornerRadius = UDim.new(0, r)
    return v
end

local function addStroke(p, c, t, tr)
    local v = Instance.new("UIStroke", p)
    v.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    v.Color = c
    v.Thickness = t or 1
    v.Transparency = tr or 0
    return v
end

local dragging = false
local dragStart
local startPos

Header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Window.Position
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        Window.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

local function StopAuto()
    if autoTask then
        task.cancel(autoTask)
        autoTask = nil
    end
    isRunning = false
    StartBtn.Text = "▶"
    StartBtn.BackgroundColor3 = COLORS.Green
    StatusLabel.Text = "⏸"
    StatusLabel.TextColor3 = COLORS.Dim
end

local function StartAuto()
    if not selectedCase then
        StatusLabel.Text = "❌"
        StatusLabel.TextColor3 = COLORS.Red
        task.wait(2)
        StatusLabel.Text = "⏸"
        StatusLabel.TextColor3 = COLORS.Dim
        return
    end
    
    if isRunning then
        StopAuto()
        return
    end
    
    isRunning = true
    StartBtn.Text = "■"
    StartBtn.BackgroundColor3 = COLORS.Red
    StatusLabel.Text = "▶"
    StatusLabel.TextColor3 = COLORS.Green

    autoTask = task.spawn(function()
        while isRunning do
            pcall(function()
                OpenCase(selectedCase, openCount)
            end)
            task.wait(0.1) -- Задержка 0.1 секунды
        end
    end)
end

StartBtn.MouseButton1Click:Connect(StartAuto)
